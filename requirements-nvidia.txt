# NVIDIA 顯卡（CUDA）用：pip install --timeout 20 --retries 10 -r requirements-nvidia.txt
# 給學校 RTX 5080 Laptop 16GB（Blackwell 架構）。Blackwell **必須用 CUDA 12.8 版 torch**，
# 舊的 cu121/cu118 版不認得這張卡（會報 "no kernel image is available"）。
# torch 版本跟 AMD 那台一樣釘 2.9.1，讓兩台行為一致。
-r requirements.txt

--extra-index-url https://download.pytorch.org/whl/cu128
torch==2.9.1+cu128

# 只有小顯存（< 10GB）的 NVIDIA 機器才會用到 4-bit；16GB 機器 auto 會走 bf16，
# 裝著不影響。0.45 起支援 Blackwell。
bitsandbytes==0.48.2
